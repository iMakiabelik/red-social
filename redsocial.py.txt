def mostrar_perfil(perfil):
    print(f"Usuario: {perfil['nombre']} | Edad: {perfil['edad']} | Ciudad: {perfil['ciudad']}")
    print("\nAmigos:")
    for amigo in perfil['amigos']:
        print(f" - {amigo['nombre']} (Amigos desde hace {amigo['tiempo_amigos']})")
    print("\nPosts:")
    for post in perfil['posts']:
        print(f" - '{post['contenido']}' | Likes: {post['likes']} | Comentarios: {len(post['comentarios'])}")
    print("-" * 50)

def ejercicio_1():
    print("=== EJERCICIO 1: SISTEMA DE RED SOCIAL ===\n")
    
    perfil_usuario = {
        "nombre": "Carlos",
        "edad": 28,
        "ciudad": "Barcelona",
        "amigos": [
            {"nombre": "Ana", "tiempo_amigos": "2 años"},
            {"nombre": "Luis", "tiempo_amigos": "6 meses"},
            {"nombre": "Sofia", "tiempo_amigos": "5 años"}
        ],
        "posts": [
            {
                "contenido": "¡Hola a todos en la red social!",
                "likes": 12,
                "comentarios": ["¡Bienvenido!", "Qué genial"]
            }
        ]
    }

    print("Estado inicial:")
    mostrar_perfil(perfil_usuario)

    nuevo_post = {
        "contenido": "Disfrutando de un café por la tarde.",
        "likes": 5,
        "comentarios": ["¡Invita!"]
    }
    perfil_usuario["posts"].append(nuevo_post)

    perfil_usuario["ciudad"] = "Madrid"

    if len(perfil_usuario["amigos"]) >= 2:
        amigo_eliminado = perfil_usuario["amigos"].pop(1)
        print(f"Se ha eliminado a {amigo_eliminado['nombre']} de la lista de amigos.\n")

    print("Estado actualizado:")
    mostrar_perfil(perfil_usuario)


def buscar_pedidos_por_email(pedidos, email_cliente):
    pedidos_encontrados = []
    for pedido in pedidos:
        if pedido["cliente"]["email"] == email_cliente:
            pedidos_encontrados.append(pedido)
    return pedidos_encontrados

def listar_estudiantes_en_curso(estudiantes, id_curso):
    matriculados = []
    for est in estudiantes:
        for curso in est["cursos_matriculados"]:
            if curso["id_curso"] == id_curso:
                matriculados.append(est["nombre"])
                break
    return matriculados
